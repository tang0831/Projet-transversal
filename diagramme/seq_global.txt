sequenceDiagram
    actor A as Agent de Saisie
    participant UC as UtilisateurController
    participant CC as CitoyenController
    participant AC as ActeController
    participant DC as DemandeController
    participant PDF as PDFGenerator
    participant DB as Base de Données

    %% 1. Authentification
    rect rgb(240, 245, 255)
        note right of A: Scénario: Connexion au système
        A->>UC: login(nom, mot_de_passe)
        UC->>DB: verifier_identifiants(nom)
        DB-->>UC: donnees_utilisateur (id_localite, role, hachage)
        UC-->>A: acces_autorise (Token + restriction géographique)
    end

    %% 2. Gestion Citoyens et Actes (Filtrés par localité de l'Agent)
    rect rgb(240, 255, 240)
        note right of A: Scénario: Enregistrement Civil Local
        A->>CC: create_citoyen(donnees)
        CC->>DB: inserer_citoyen(donnees, id_localite)
        DB-->>CC: confirmation_insertion
        CC-->>A: succes_creation_citoyen

        A->>AC: ajouter_acte(id_citoyen, type_acte)
        AC->>DB: inserer_acte(id_citoyen, type_acte, id_localite)
        DB-->>AC: confirmation_insertion
        AC-->>A: succes_creation_acte
    end

    %% 3. Demandes d'actes et Génération PDF
    rect rgb(255, 245, 240)
        note right of A: Scénario: Traitement et Impression
        A->>DC: creer_demande(id_utilisateur, type_acte)
        DC->>DB: enregistrer_demande(demande)
        DB-->>DC: confirmation_demande
        DC-->>A: demande_creee

        A->>AC: generer_pdf_acte(id_acte)
        AC->>PDF: generer(donnees_acte)
        PDF-->>AC: fichier_pdf (binaire)
        AC-->>A: telechargement_pdf
    end

    %% 4. Forum / Communication inter-districts
    rect rgb(255, 255, 240)
        note right of A: Scénario: Publication d'une annonce/note locale
        A->>DC: ajouter_message(message, district)
        DC->>DB: inserer_message(message, district)
        DB-->>DC: confirmation_publication
        DC-->>A: message_publie
    end
